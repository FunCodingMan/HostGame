void AnimationController::Setup(const sf::Texture& tex, int frameW, int frameH)
{
    m_texture = &tex;
    m_frameW  = frameW;
    m_frameH  = frameH;
    m_nextRow = 0;
}

void AnimationController::AddClip(const std::string& name, int frameCount, float frameTime)
{
    if (m_texture == nullptr || frameCount <= 0) return;

    int rows = (int)(m_texture->getSize().y / m_frameH);
    if (m_nextRow >= rows) return;

    int row = m_nextRow++;
    AddClipRow(name, row, 0, frameCount, frameTime, true);
}

void AnimationController::AddClip(const std::string& name, int frameCount, float frameTime,
                                  int row, int startCol, bool loop)
{
    if (m_texture == nullptr || frameCount <= 0 || row < 0 || startCol < 0) return;

    int rows = (int)(m_texture->getSize().y / m_frameH);
    if (row >= rows) return;

    AddClipRow(name, row, startCol, frameCount, frameTime, loop);

    if (row + 1 > m_nextRow) m_nextRow = row + 1;
}

void AnimationController::AddClip(const std::string& name,
                                  const std::vector<sf::IntRect>& frames,
                                  float frameTime, bool loop)
{
    m_clips[name] = Clip{ frames, frameTime, loop };
}

void AnimationController::AddClipRow(const std::string& name, int row, int startCol,
                                     int count, float frameTime, bool loop)
{
    int cols = (int)(m_texture->getSize().x / m_frameW);
    int realCount = std::min(count, cols - startCol);
    if (realCount <= 0) return;

    std::vector<sf::IntRect> frames;
    frames.reserve(realCount);
    for (int i = 0; i < realCount; ++i)
    {
        frames.emplace_back((startCol + i) * m_frameW, row * m_frameH, m_frameW, m_frameH);
    }

    bool  actualLoop = loop && (realCount > 1);
    float ft         = (realCount > 1) ? frameTime : 1.f;

    m_clips[name] = Clip{ frames, ft, actualLoop };
}

void AnimationController::Play(const std::string& name, bool restart)
{
    if (m_current == name && !restart) return;
    if (!m_clips.count(name)) return;
    m_current = name;
    m_frame   = 0;
    m_elapsed = 0.f;
}

bool AnimationController::IsFinished() const
{
    auto it = m_clips.find(m_current);
    if (it == m_clips.end() || it->second.frames.empty()) return true;
    return !it->second.loop && m_frame >= (int)it->second.frames.size() - 1;
}

void AnimationController::Update(float dt)
{
    auto it = m_clips.find(m_current);
    if (it == m_clips.end() || it->second.frames.empty()) return;

    const Clip& c = it->second;
    m_elapsed += dt;
    while (m_elapsed >= c.frameTime)
    {
        m_elapsed -= c.frameTime;
        if (++m_frame >= (int)c.frames.size())
            m_frame = c.loop ? 0 : (int)c.frames.size() - 1;
    }
}

void AnimationController::Apply(sf::Sprite& sprite) const
{
    auto it = m_clips.find(m_current);
    if (it == m_clips.end() || it->second.frames.empty()) return;
    sprite.setTextureRect(it->second.frames[m_frame]);
}

Camera::Camera(sf::Vector2f size) : m_lerpSpeed(CAMERA_LERP_FACTOR)
{
    m_view.setSize(size);
    m_view.setCenter(size.x / 2.f, size.y / 2.f);
    m_deadZone = sf::Vector2f(CAMERA_DEAD_ZONE_WIDTH, CAMERA_DEAD_ZONE_HEIGHT);

    m_currentFocus = m_view.getCenter();
    m_currentMouseOffset = sf::Vector2f(0.f, 0.f);
}

void Camera::Update(sf::Vector2f targetPos, sf::Vector2f targetMouseOffset, float dt, float mapWidthPx, float mapHeightPx)
{
    sf::Vector2f desiredFocus = m_currentFocus;
    
    if (targetPos.x > m_currentFocus.x + m_deadZone.x) {
        desiredFocus.x = targetPos.x - m_deadZone.x;
    } else if (targetPos.x < m_currentFocus.x - m_deadZone.x) {
        desiredFocus.x = targetPos.x + m_deadZone.x;
    }

    if (targetPos.y > m_currentFocus.y + m_deadZone.y) {
        desiredFocus.y = targetPos.y - m_deadZone.y;
    } else if (targetPos.y < m_currentFocus.y - m_deadZone.y) {
        desiredFocus.y = targetPos.y + m_deadZone.y;
    }

    m_currentFocus += (desiredFocus - m_currentFocus) * m_lerpSpeed * dt;

    m_currentMouseOffset += (targetMouseOffset - m_currentMouseOffset) * m_lerpSpeed * dt;

    sf::Vector2f finalPos = m_currentFocus + m_currentMouseOffset;
    
    float halfWidth = m_view.getSize().x / 2.f;
    float halfHeight = m_view.getSize().y / 2.f;

    finalPos.x = std::clamp(finalPos.x, halfWidth, mapWidthPx - halfWidth);
    finalPos.y = std::clamp(finalPos.y, halfHeight, mapHeightPx - halfHeight);

    m_view.setCenter(finalPos);
}

const sf::View& Camera::GetView() const
{
    return m_view;
}

void Camera::SetSize(sf::Vector2f newSize)
{
    m_view.setSize(newSize);
}

void Camera::Zoom(float factor)
{
    m_view.zoom(factor);
}

    {
        sf::View oldView = window.getView();
        window.setView(worldView);

        std::vector<sf::Vertex> vertices;
        vertices.reserve(s_lines.size() * 2);
        for (const auto &line : s_lines)
        {
            vertices.push_back(sf::Vertex(line.p1, line.color));
            vertices.push_back(sf::Vertex(line.p2, line.color));
        }
        window.draw(vertices.data(), vertices.size(), sf::Lines);
        s_lines.clear(); // Сбрасываем линии после рендера текущего кадра

        window.setView(oldView);
    }

    if (!s_show)
        return;
    if (s_currentFont == nullptr)
        return;

    sf::Vector2i pixel = sf::Mouse::getPosition(window);
    sf::Vector2f world = window.mapPixelToCoords(pixel, worldView);

    int tileX = static_cast<int>(std::floor(world.x / Config::WORLD_TILE_SIZE));
    int tileY = static_cast<int>(std::floor(world.y / Config::WORLD_TILE_SIZE));

    std::ostringstream ss;
    ss << "Px: (" << pixel.x << ", " << pixel.y << ")   " << "World: (" << std::fixed
       << std::setprecision(1) << world.x << ", " << world.y << ")   " << "Tile: (" << tileX << ", "
       << tileY << ")";

    s_text.setString(StringUtils::ToSfStringFromUtf8(ss.str()));
    window.draw(s_text);
}

void Game::Run() 
{
    sf::Clock clock;
    float accumulator = 0.0f;

    while (m_data->window.isOpen()) 
    {
        m_data->manager.ProcessStateChanges();

        ProcessEvents();

        float frameTime = clock.restart().asSeconds();
        if (frameTime > 0.25f) frameTime = 0.25f;
        accumulator += frameTime;

        while (accumulator >= m_dt) 
        {
            Update();
            accumulator -= m_dt;
        }

        Render();
    }
}

void Game::ProcessEvents()
{
    sf::Event event;
    while (m_data->window.pollEvent(event)) 
    {
        if (event.type == sf::Event::Closed) {
            m_data->window.close();
        }
        
        if (!m_data->manager.isEmpty()) {
            m_data->manager.GetActiveState()->HandleInput(event);
        }
    }
}

void Game::Update()
{
    if (!m_data->manager.isEmpty()) {
        m_data->manager.GetActiveState()->Update(m_dt);
    }
}

void Game::Render()
{
    m_data->window.clear(Config::BG_COLOR);
    
    if (!m_data->manager.isEmpty()) {
        m_data->manager.GetActiveState()->Draw();
    }

    m_data->window.display();
}

bool LevelLoader::IsBoundsBlocked(const Map& map, const sf::FloatRect& bounds)
{
    const float ts = static_cast<float>(Config::WORLD_TILE_SIZE);

    int minX = static_cast<int>(std::floor(bounds.left / ts));
    int maxX = static_cast<int>(std::floor((bounds.left + bounds.width)  / ts));
    int minY = static_cast<int>(std::floor(bounds.top / ts));
    int maxY = static_cast<int>(std::floor((bounds.top  + bounds.height) / ts));

    for (int ty = minY; ty <= maxY; ++ty)
    {
        for (int tx = minX; tx <= maxX; ++tx)
        {
            if (map.IsSolid(tx, ty, Config::EntityType::Parasite))
            {
                return true;
            }
        }
    }

    return false;
}

void LevelLoader::LoadEnemies(const std::string& path,
                              std::vector<std::unique_ptr<Host>>& out,
                              const Map& map)
{
    std::ifstream file(path);
    if (!file.is_open()) {
        std::cerr << "[LevelLoader] Cannot open enemies file '" << path << "'\n";
        return;
    }

    std::string line;
    int lineNum = 0;
    while (std::getline(file, line)) {
        ++lineNum;
        if (line.empty() || line[0] == '#') continue;

        std::istringstream ss(line);
        std::string type;
        float x, y;
        if (!(ss >> type >> x >> y))
        {
            std::cerr << "[LevelLoader] " << path << ":" << lineNum
                      << " invalid format (expected 'Type x y [decay]')\n";
            continue;
        }

        sf::Vector2f pos(x, y);
        std::unique_ptr<Host> enemy;
        if (type == "Scientist")
        {
            enemy = std::make_unique<Scientist>(pos);
        } 
        else if (type == "Military")
        {
            enemy = std::make_unique<Military>(pos);
        }
        else
        {
            std::cerr << "[LevelLoader] " << path << ":" << lineNum
                      << " unknown enemy type '" << type << "'\n";
            continue;
        }

        float customDecay = 0.f;
        if (ss >> customDecay) enemy->SetMaxDecayTime(customDecay);

        if (IsBoundsBlocked(map, enemy->GetBounds()))
        {
            std::cerr << "[LevelLoader] " << path << ":" << lineNum
                      << " cannot spawn " << type << " at (" << x << "," << y
                      << "): hitbox overlaps a solid tile\n";
            continue;
        }

        out.push_back(std::move(enemy));
    }
}

void Level::UpdateCamera(float dt, const LevelInput& input)
{
    sf::Vector2f targetPos = m_playerAvatar->GetPosition();
    sf::Vector2i mousePixelPos = input.mousePixelPos;
    sf::Vector2u windowSize = input.windowSize;

    float normX = (mousePixelPos.x - (windowSize.x / 2.f)) / (windowSize.x / 2.f);
    float normY = (mousePixelPos.y - (windowSize.y / 2.f)) / (windowSize.y / 2.f);

    normX = std::clamp(normX, -1.f, 1.f);
    normY = std::clamp(normY, -1.f, 1.f);

    sf::Vector2f viewSize = m_camera.GetView().getSize();
    float maxOffsetX = (viewSize.x / 2.f) * Config::CAMERA_LOOK_STRENGTH;
    float maxOffsetY = (viewSize.y / 2.f) * Config::CAMERA_LOOK_STRENGTH;

    sf::Vector2f mouseOffset(normX * maxOffsetX, normY * maxOffsetY);

    m_camera.Update(targetPos, mouseOffset, dt,
                    m_map.GetWidthInPixels(), m_map.GetHeightInPixels());
}

sf::Vector2f Level::MapMouseToWorld(const LevelInput& input) const
{
    const sf::View& view = m_camera.GetView();
    float w = static_cast<float>(input.windowSize.x);
    float h = static_cast<float>(input.windowSize.y);

    return sf::Vector2f(
        view.getCenter().x - view.getSize().x / 2.f + (input.mousePixelPos.x / w) * view.getSize().x,
        view.getCenter().y - view.getSize().y / 2.f + (input.mousePixelPos.y / h) * view.getSize().y);
}

void Level::Update(float dt, const LevelInput& input)
{
    m_stats.timeSpent += dt;

    m_map.Update(dt);

    UpdateCamera(dt, input);

    GameContext ctx;
    ctx.dt = dt;
    ctx.mousePos = MapMouseToWorld(input);
    ctx.levelMap = &m_map;
    ctx.audio = m_audio;
    ctx.playerAvatar = m_playerAvatar;
    ctx.avatarIsParasite = (m_playerAvatar == &m_parasite);
    ctx.bullets = &m_bullets;

    m_parasite.Update(ctx);

    for (auto& enemy : m_hosts)
    {
        enemy->Update(ctx);
    }
    for (auto& syringe : m_syringes)
    {
        syringe->Update(ctx);
    }
    
    const std::vector<sf::Vector2i> brokenTiles = m_bullets.Update(dt, m_map);

    if (!brokenTiles.empty())
    {
        m_audio->PlaySound(SOUND_ID_BREAK_GLASS);
    }

    HandleBullets();
    HandlePossession();
    HandleEjection(ctx);
    HandleItems(ctx);
    UpdateHeartbeat();

    CleanDeadEnemies();

    m_syringes.erase(std::remove_if(m_syringes.begin(), m_syringes.end(),
                                    [](const auto& s) { return s->IsCollected(); }),
                     m_syringes.end());

    CheckEndConditions();
}

void Level::HandleBullets()
{
    for (Bullet& bullet : m_bullets.GetBullets())
    {
        if (!bullet.active)
        {
            continue;
        }

        sf::FloatRect bulletBounds = BulletManager::GetBulletBounds(bullet);

        if (!bullet.fromPlayer)
        {
            if (m_playerAvatar != nullptr && !m_playerAvatar->IsDead() &&
                bulletBounds.intersects(m_playerAvatar->GetBounds()))
            {
                m_playerAvatar->TakeDamage(bullet.damage);
                bullet.active = false;

                if (!m_playerAvatar->IsDead())
                {
                    const char* sound = m_playerAvatar->GetDamageSound();
                    if (sound != nullptr && m_audio->HasSound(sound))
                    {
                        m_audio->PlaySound(sound);
                    }
                }
            }
            continue;
        }

        for (auto& enemy : m_hosts)
        {
            if (enemy->IsDead() || enemy.get() == m_currentHost)
            {
                continue;
            }

            if (bulletBounds.intersects(enemy->GetBounds()))
            {
                bool wasDead = enemy->IsDead();
                enemy->TakeDamage(bullet.damage);
                bullet.active = false;

                if (!wasDead && enemy->IsDead())
                {
                    const char* sound = enemy->GetDeathSound();
                    if (m_audio->HasSound(sound))
                    {
                        m_audio->PlaySound(sound);
                    }
                }
                else if (!wasDead)
                {
                    const char* sound = enemy->GetDamageSound();
                    if (sound != nullptr && m_audio->HasSound(sound))
                    {
                        m_audio->PlaySound(sound);
                    }
                }
                break;
            }
        }
    }
}

void Level::HandlePossession()
{
    if (m_playerAvatar != &m_parasite || !m_parasite.IsDashing())
        return;

    for (auto& enemy : m_hosts)
    {
        if (!enemy->IsDead() && m_parasite.GetBounds().intersects(enemy->GetBounds()))
        {
            m_parasite.SetPlayerControlled(false);
            m_parasite.ResetMovement();

            enemy->SetPlayerControlled(true);
            enemy->ResetDecayTimer();
            enemy->OnPossessed();

            m_playerAvatar = enemy.get();
            m_currentHost = enemy.get();
            m_timeInCurrentHost = 0.f;

            m_stats.hostsPossessed++;
            m_audio->PlaySound(SOUND_ID_HOST);
            break;
        }
    }
}

void Level::HandleEjection(const GameContext& ctx)
{
    if (m_currentHost == nullptr || m_playerAvatar != m_currentHost)
        return;

    m_timeInCurrentHost += ctx.dt;

    bool manualEject = GameKeyboard::IsEjectKeyPressed();
    bool timeOut = (m_currentHost->GetDecayTimer() <= 0.f);
    bool hostDead = m_currentHost->IsDead();

    if (manualEject && m_timeInCurrentHost < Config::COOLDOWN_IN_HOST)
        manualEject = false;

    if (manualEject || timeOut || hostDead)
    {
        sf::Vector2f ejectPosition = m_currentHost->GetPosition();
        bool hostDiedNow = !hostDead && (manualEject || timeOut);

        m_currentHost->SetPlayerControlled(false);
        m_currentHost->OnEjected();

        if (hostDiedNow)
        {
            m_currentHost->Kill();
        }

        if (hostDiedNow || hostDead)
        {
            const char* sound = m_currentHost->GetDeathSound();
            if (ctx.audio->HasSound(sound))
            {
                ctx.audio->PlaySound(sound);
            }
        }

        m_parasite.SetPosition(ejectPosition);
        m_parasite.ResetMovement();
        m_parasite.SetPlayerControlled(true);

        if (manualEject)
            m_parasite.ForceDash(ctx);

        m_playerAvatar = &m_parasite;
        m_currentHost = nullptr;
        m_timeInCurrentHost = 0.f;
    }
}

void Level::HandleItems(const GameContext& ctx)
{
    for (auto& syringe : m_syringes)
    {
        if (syringe->IsCollected())
            continue;

        if (m_playerAvatar->GetBounds().intersects(syringe->GetBounds()))
        {
            if (m_currentHost != nullptr && m_playerAvatar == m_currentHost)
            {
                syringe->Collect();
                m_stats.syringesCollected++;
                m_currentHost->AddDecayTime(Config::SYRINGE_DECAY_RESTORE);
                ctx.audio->PlaySound(SOUND_ID_COLLECT);
            }
        }
    }
}

void Level::CheckEndConditions()
{
    if (m_result != LevelResult::None)
        return;

    bool isParasiteDead = (m_parasite.GetLife() <= 0.f);
    bool fellIntoPit = (m_playerAvatar->GetPosition().y > m_map.GetHeightInPixels());

    if (isParasiteDead || fellIntoPit)
    {
        m_audio->StopSound(SOUND_ID_HEARTBEAT);
        m_audio->PlaySound(SOUND_ID_LOSE);
        m_audio->PlaySound(SOUND_ID_DEATH);
        m_result = LevelResult::Failed;
        return;
    }

    if (m_map.IsIntersectingWith(m_playerAvatar->GetBounds(), BlockNames::Elevator))
    {
        int access = m_playerAvatar->GetProperties().GetInt("access_level", 0);
        if (access >= 1)
        {
            m_audio->StopSound(SOUND_ID_HEARTBEAT);
            m_audio->PlaySound(SOUND_ID_WIN);
            m_result = LevelResult::Completed;
        }
    }
}

bool Map::LoadTileConfig(const string& filename)
{
    ifstream file(filename);
    if(!file.is_open()) return false;

    string line;

    while (getline(file, line))
    {
        if (line.empty() || line[0] == '#')
        {
            continue;
        }
        stringstream ss(line);
        string code, collisionStr;

        int col, row;

        if (ss >> code >> collisionStr >> col >> row)
        {
            if (m_stringToBlock.find(code) == m_stringToBlock.end()) {
                std::cerr << "Внимание: Найден неизвестный код '" << code << "' в " << filename << std::endl;
                continue;
            }
            ParseTileConfig(code, collisionStr, col, row);
        }
    }
    return true;
}

void Map::LoadLeveFromLine(const string& line)
{
    stringstream ss(line);
    string tileCode;
    int curLineWidth = 0;

    while (ss >> tileCode)
    {
        if (m_stringToBlock.find(tileCode) == m_stringToBlock.end())
        {
            std::cerr << "Предупреждение: Неизвестный тайл '" << tileCode << "'" << std::endl;
            m_grid.push_back(Config::BlockNames::Empty);
        }
        else
        {
            m_grid.push_back(m_stringToBlock[tileCode]);
        }
        curLineWidth++;
    }
    if (m_width == 0) 
    {
        m_width = curLineWidth;
    }
    m_height++;
}

bool Map::LoadFromFile(const string& filename)
{
    ifstream file(filename);

    if (!file.is_open())
    {
        std::cerr << "Ошибка: Не удалось загрузить карту " << filename << std::endl;
        return false;
    }

    m_grid.clear();
    m_doors.clear();

    string line;
    m_height = 0;
    m_width = 0;

    while (getline(file, line))
    {
        if (line.empty())
        {
            continue;
        }

        LoadLeveFromLine(line);
    }

    for (int i = 0; i < static_cast<int>(m_grid.size()); ++i)
    {
        if (m_grid[i] == Config::BlockNames::Door)
        {
            m_doors.emplace_back(i % m_width, i / m_width, m_doorTexture);
        }
    }

    BuildVertexArray();

    return true;
}

void Map::Update(float dt)
{
    for (Door& door : m_doors)
    {
        door.Update(dt);

        if (!door.IsOpen())
        {
            continue;
        }

        int index = door.GetTileY() * m_width + door.GetTileX();

        if (m_grid[index] == Config::BlockNames::Door)
        {
            m_grid[index] = Config::BlockNames::Empty;
            BuildVertexElement(door.GetTileX(), door.GetTileY());
        }
    }
}

bool Map::IsSolid(int x, int y, Config::EntityType entityType) const 
{
    Config::BlockNames block = GetTile(x, y);
    const Config::TileData* data = GetTileData(block);
    
    if (!data) return true; 

    Config::CollisionType type = data->collision;

    if (type == Config::CollisionType::Solid || type == Config::CollisionType::Breakable) return true;
    if (type == Config::CollisionType::Empty) return false;
    
    if (type == Config::CollisionType::Ventilation) 
    {
        return entityType != Config::EntityType::Parasite; 
    }

    if (type == Config::CollisionType::Door)
    {
        return true;
    }

    return false;
}

void Map::BuildVertexElement(int xTile, int yTile)
{
    Config::BlockNames blockName = GetTile(xTile, yTile);

    const Config::TileData* tileData = GetTileData(blockName);
    if (!tileData)
    {
        return;
    }

    int vertexElIdx = (xTile + yTile * m_width) * 4;

    sf::Vertex* quad = &m_Vertices[vertexElIdx];
    float wSize = Config::WORLD_TILE_SIZE;
    float x = xTile * wSize;
    float y = yTile * wSize;

    quad[0].position = sf::Vector2f(x, y);
    quad[1].position = sf::Vector2f(x + wSize, y);
    quad[2].position = sf::Vector2f(x + wSize, y + wSize);
    quad[3].position = sf::Vector2f(x, y + wSize);

    sf::IntRect texRect = tileData->textureRect;
    float tx = texRect.left;
    float ty = texRect.top;
    float tSize = texRect.width;

    quad[0].texCoords = sf::Vector2f(tx, ty);
    quad[1].texCoords = sf::Vector2f(tx + tSize, ty);
    quad[2].texCoords = sf::Vector2f(tx + tSize, ty + tSize);
    quad[3].texCoords = sf::Vector2f(tx, ty + tSize);
}

void Map::BuildVertexArray()
{
    m_Vertices.resize(m_width * m_height * 4);

    for (int i = 0; i < m_width; ++i)
    {
        for (int j = 0; j < m_height; ++j)
        {
            BuildVertexElement(i, j);
        }
    }
}

bool Map::IsIntersectingWith(const sf::FloatRect& bounds, Config::BlockNames targetBlock) const
{
    int minX = std::floor(bounds.left / Config::WORLD_TILE_SIZE);
    int maxX = std::floor((bounds.left + bounds.width) / Config::WORLD_TILE_SIZE);
    int minY = std::floor(bounds.top / Config::WORLD_TILE_SIZE);
    int maxY = std::floor((bounds.top + bounds.height) / Config::WORLD_TILE_SIZE);

    for (int y = minY; y <= maxY; ++y)
    {
        for (int x = minX; x <= maxX; ++x)
        {
            if (GetTile(x, y) == targetBlock)
            {
                return true;
            }
        }
    }
    return false;
}

bool SoundManager::LoadSound(const std::string& id, const std::string& filepath, float volume)
{
    auto buffer = std::make_unique<sf::SoundBuffer>();
    if (!buffer->loadFromFile(filepath))
    {
        std::cerr << "Audio Error: Could not load sound '" << filepath << "'\n";
        return false;
    }

    sf::Sound sound;
    sound.setBuffer(*buffer);
    sound.setVolume(volume);

    m_soundBuffers[id] = std::move(buffer);
    m_sounds[id] = sound;
    
    return true;
}

void SoundManager::PlaySound(const std::string& id, bool loop)
{
    auto it = m_sounds.find(id);
    if (it != m_sounds.end())
    {
        it->second.setLoop(loop);

        if (loop && it->second.getStatus() == sf::Sound::Playing)
        {
            return;

bool Raycast::IsWithinFov(sf::Vector2f dirToTarget, sf::Vector2f faceDir, float fovCosine)
{
    float faceLen = std::hypot(faceDir.x, faceDir.y);
    if (faceLen == 0.f)
    {
        return true;
    }

    sf::Vector2f unitFace = faceDir / faceLen;
    float dot = dirToTarget.x * unitFace.x + dirToTarget.y * unitFace.y;
    return dot >= fovCosine;
}

bool Raycast::CanSee(const Map& map,
                     sf::Vector2f eye,
                     sf::Vector2f target,
                     float maxRange,
                     float fovCosine,
                     sf::Vector2f faceDir)
{
    sf::Vector2f toTarget = target - eye;
    float distance = std::hypot(toTarget.x, toTarget.y);

    if (distance > maxRange)
    {
        return false;
    }

    if (distance == 0.f)
    {
        return true;
    }

    sf::Vector2f dir = toTarget / distance;
    if (!IsWithinFov(dir, faceDir, fovCosine))
    {
        return false;
    }

    float tileSize = static_cast<float>(map.GetTileSize());

    for (float walked = 0.f; walked < distance; walked += Config::LOS_STEP_SIZE)
    {
        sf::Vector2f point = eye + dir * walked;
        int tileX = static_cast<int>(std::floor(point.x / tileSize));
        int tileY = static_cast<int>(std::floor(point.y / tileSize));

        if (map.BlocksSight(tileX, tileY))
        {
            DrawDebugRay(eye, point, false);
            return false;
        }
    }

    DrawDebugRay(eye, target, true);
    return true;
}

void BulletManager::Spawn(sf::Vector2f origin, sf::Vector2f direction, float damage,
                          bool fromPlayer)
{
    float length = std::hypot(direction.x, direction.y);
    if (length == 0.f)
    {
        return;
    }

    sf::Vector2f unitDir = direction / length;

    for (Bullet &bullet : m_pool)
    {
        if (bullet.active)
        {
            continue;
        }

        bullet.active = true;
        bullet.fromPlayer = fromPlayer;
        bullet.position = origin;
        bullet.velocity = unitDir * Config::BULLET_SPEED;
        bullet.angleDeg = std::atan2(unitDir.y, unitDir.x) * Config::RAD_TO_DEG;
        bullet.damage = damage;
        bullet.lifeTimer = Config::BULLET_MAX_LIFETIME;
        return;
    }
}

bool BulletManager::IsOutOfMap(sf::Vector2f pos, const Map &map) const

           pos.y > map.GetHeightInPixels();

std::vector<sf::Vector2i> BulletManager::Update(float dt, Map &map)
{
    std::vector<sf::Vector2i> brokenTiles;
    float tileSize = static_cast<float>(map.GetTileSize());

    for (Bullet &bullet : m_pool)
    {
        if (!bullet.active)
        {
            continue;
        }

        bullet.position += bullet.velocity * dt;
        bullet.lifeTimer -= dt;

        if (bullet.lifeTimer <= 0.f || IsOutOfMap(bullet.position, map))
        {
            bullet.active = false;
            continue;
        }

        int tileX = static_cast<int>(std::floor(bullet.position.x / tileSize));
        int tileY = static_cast<int>(std::floor(bullet.position.y / tileSize));

        if (!map.IsSolid(tileX, tileY, Config::EntityType::Bullet))
        {
            continue;
        }

        if (map.IsBreakable(tileX, tileY))
        {
            map.BreakTile(tileX, tileY);
            brokenTiles.emplace_back(tileX, tileY);
        }

        bullet.active = false;
    }

    return brokenTiles;
}

bool Character::CheckMoveKeys(float dt)

void Character::ResolveFriction(float dt)
{
    float f = m_isOnGround ? m_friction : m_friction * 0.3f;

    if (m_velocity.x > 0.f)
    {
        m_velocity.x -= f * dt;

        if (m_velocity.x < 0.f)
        {
            m_velocity.x = 0.f;
        }
    }
    else
    {
        m_velocity.x += f * dt;

        if (m_velocity.x > 0.f)
        {
            m_velocity.x = 0.f;
        }
    }
}

        {
            if (const char* sound = GetLandSound(); sound && ctx.audio->HasSound(sound)) {
                ctx.audio->PlaySound(sound); 
            }
        }

        OnLanding();
    }
    else if (m_velocity.y < 0.f)
    {
        m_hitbox.setPosition(m_hitbox.getPosition().x,
                             (y + 1) * tileSize + (bounds.height / 2.f) + Config::EPSILON);
        m_velocity.y = 0.f;
    }
}

bool Character::TryOpenDoor(int x, int y, const GameContext& ctx)
{
    if (!m_isPlayerControlled) return false;
    if (!ctx.levelMap->IsDoor(x, y)) return false;
    if (m_properties.GetInt("access_level", 0) < 1) return false;

    if (ctx.levelMap->OpenDoor(x, y) && ctx.audio->HasSound(Config::SOUND_ID_DOOR_OPEN))
    {
        ctx.audio->PlaySound(Config::SOUND_ID_DOOR_OPEN);
    }

    return false;
}

void Character::ResolveCollisions(const GameContext& ctx, Axis axis)
{
    sf::FloatRect bounds = m_hitbox.getGlobalBounds();
    int tileSize = ctx.levelMap->GetTileSize();

    int minX = (int)std::floor(bounds.left / tileSize);
    int maxX = (int)std::floor((bounds.left + bounds.width)  / tileSize);
    int minY = (int)std::floor(bounds.top  / tileSize);
    int maxY = (int)std::floor((bounds.top + bounds.height) / tileSize);

    if (axis == Axis::Vertical)
    {
        m_isOnGround = false;
    }

    for (int y = minY; y <= maxY; ++y)
    {
        for (int x = minX; x <= maxX; ++x)
        {
            if (!ctx.levelMap->IsSolid(x, y, GetEntityType()))
            {
                continue;
            }

            if (TryBreakTile(x, y, ctx))
            {
                continue;
            }

            if (TryOpenDoor(x, y, ctx))
            {
                continue;
            }

            if (axis == Axis::Horizontal)
            {
                ResolveCollisionsX(x, ctx);
            }
            else
            {
                ResolveCollisionsY(y, ctx);
            }
        }
    }
}

void Character::UpdateSpriteDirection()
{
    m_sprite.setScale(m_lookDir < 0 ? -m_scale : m_scale, m_scale);
}

void Character::UpdateAnimation(float dt)
{
    m_anim.Update(dt);
    m_anim.Apply(m_sprite);
}

void Character::ApplyMovement(const GameContext& ctx)
{
    m_hitbox.move(m_velocity.x * ctx.dt, 0.f);
    ResolveCollisions(ctx, Axis::Horizontal);

    m_hitbox.move(0.f, m_velocity.y * ctx.dt);
    ResolveCollisions(ctx, Axis::Vertical);

    UpdateSpriteDirection();
    UpdateAnimation(ctx.dt);

    if (m_isOnGround && std::abs(m_velocity.x) > Config::WALK_SOUND_VELOCITY) 
    {
        m_stepTimer -= ctx.dt;
        if (m_stepTimer <= 0.f)
        {
            if (const char* sound = GetWalkSound(); sound && ctx.audio->HasSound(sound)) {
                ctx.audio->PlaySound(sound); 
            }
            m_stepTimer = m_stepSoundInterval;
        }
    }
    else
    {
        m_stepTimer = 0.f;
    }

    sf::Vector2f eff = m_spriteOffset;

    if (m_sprite.getScale().x < 0.f)
    {
        eff.x = -eff.x;
    }

    m_sprite.setPosition(m_hitbox.getPosition() + eff);
}

    if (state == EntityState::Dead ||
        state == EntityState::Alert ||
        state == EntityState::Attack ||
        state == EntityState::Reload ||
        state == EntityState::Report)
    {

    if (m_isOnGround)
    {
        if (std::abs(m_velocity.x) > Config::HOST_WALK_THRESHOLD)

void Military::UpdatePatrol(const GameContext& ctx)
{
    if (!m_isOnGround)
    {
        m_velocity.x = 0.f;
        return;
    }

    const float tileSize = static_cast<float>(ctx.levelMap->GetTileSize());
    sf::FloatRect bounds = GetBounds();

    auto isSolidAt = [&](float x, float y)
    {
        int tx = static_cast<int>(std::floor(x / tileSize));
        int ty = static_cast<int>(std::floor(y / tileSize));
        return ctx.levelMap->IsSolid(tx, ty, Config::EntityType::Military);
    };

    float aheadX = m_patrolDir > 0.f ? bounds.left + bounds.width + 4.f
                                    : bounds.left - 4.f;
    bool wallAhead   = isSolidAt(aheadX, bounds.top + bounds.height * 0.5f);
    bool groundAhead = isSolidAt(aheadX, bounds.top + bounds.height + 4.f);

    if (wallAhead || !groundAhead)
    {
        m_patrolDir = -m_patrolDir;
    }

    m_velocity.x = m_patrolDir * Config::MILITARY_PATROL_SPEED;
    SetLookDir(m_patrolDir > 0.f ? 1 : -1);
}

void Military::UpdateTimers(const GameContext& ctx)
{
    m_weaponAnim.Update(ctx.dt);

    if (m_weaponAnim.Current() == "shoot" && m_weaponAnim.IsFinished())
    {
        m_weaponAnim.Play("idle");
    }

    if (m_fireTimer > 0.f)
    {
        m_fireTimer -= ctx.dt;
    }

    if (m_reactionTimer > 0.f)
    {
        m_reactionTimer -= ctx.dt;
    }

    if (m_attackTimer > 0.f)
    {
        m_attackTimer -= ctx.dt;
        if (m_attackTimer <= 0.f && m_stateManager.GetState() == EntityState::Attack)
        {
            m_stateManager.SetState(EntityState::Idle);
        }
    }

    if (m_alertTimer > 0.f)
    {
        m_alertTimer -= ctx.dt;
        if (m_alertTimer <= 0.f && m_stateManager.GetState() == EntityState::Alert)
        {
            m_stateManager.SetState(EntityState::Idle);
        }
    }

    if (m_isReloading)
    {
        m_reloadTimer -= ctx.dt;
        if (m_reloadTimer <= 0.f)
        {
            m_isReloading = false;
            m_ammo = Config::MILITARY_AMMO;

            if (m_stateManager.GetState() == EntityState::Reload)
            {
                m_stateManager.SetState(EntityState::Idle);
            }
        }
    }
}

void Military::UpdateWeaponAim(sf::Vector2f point)
{
    sf::Vector2f shoulder = GetShoulderPosition();
    sf::Vector2f direction = point - shoulder;

    if (std::hypot(direction.x, direction.y) > 0.001f)
    {
        m_aimAngle = std::atan2(direction.y, direction.x) * Config::RAD_TO_DEG;
    }
}

void Military::TryFire(const GameContext& ctx, bool fromPlayer)
{
    if (m_isReloading || m_fireTimer > 0.f || m_ammo <= 0 || ctx.bullets == nullptr)
    {
        return;
    }

    --m_ammo;
    m_fireTimer   = Config::MILITARY_FIRE_INTERVAL;
    m_attackTimer = Config::MILITARY_ATTACK_STATE_TIME;
    m_stateManager.SetState(EntityState::Attack);
    m_weaponAnim.Play("shoot", true);

    float radians = m_aimAngle / Config::RAD_TO_DEG;
    sf::Vector2f direction(std::cos(radians), std::sin(radians));

    ctx.bullets->Spawn(GetMuzzlePosition(), direction,
                       Config::MILITARY_SHOT_DAMAGE, fromPlayer);

    if (ctx.audio != nullptr)
    {
        const char* shots[Config::SOUND_SHOT_VARIANTS] = {
            Config::SOUND_ID_SHOT_1,
            Config::SOUND_ID_SHOT_2,
            Config::SOUND_ID_SHOT_3
        };

        const char* id = shots[RandomUtils::Int(0, Config::SOUND_SHOT_VARIANTS - 1)];
        if (ctx.audio->HasSound(id))
        {
            ctx.audio->PlaySound(id);
        }
    }
}

void Parasite::ForceDash(const GameContext& ctx)
{
    ctx.audio->PlaySound(Config::SOUND_ID_DASH);
    sf::Vector2f targetPos = ctx.mousePos;

    sf::Vector2f center = m_hitbox.getPosition();

    float dx = targetPos.x - center.x;
    float dy = targetPos.y - center.y;
    float length = std::sqrt(dx * dx + dy * dy);

    if (length == 0.f)
    {
        return;
    }

    ++m_dashCount;

    m_velocity.x = (dx / length) * m_dashForce;
    m_velocity.y = (dy / length) * m_dashForce;

    if (std::abs(dx) > 0.1f)
    {
        SetLookDir(dx > 0.f ? 1 : -1);
    }

    m_dashCooldown = Config::DASH_COOLDOWN_TIME;
    m_hasDash      = false;
    m_isOnGround   = false;

    m_stateManager.SetState(EntityState::Dash);
}

    m_velocity = sf::Vector2f(0.f, 0.f);
    m_isOnGround = false;

    m_stateManager.SetState(EntityState::Idle);

void Parasite::HorizontalDash(float dt)
{
    float drag = m_friction * 3.f;

    if (m_velocity.x > 0.f)
    {
        m_velocity.x -= drag * dt;
        m_velocity.x = std::max(m_velocity.x, Config::PARASITE_MAX_SPEED);
    }
    else
    {
        m_velocity.x += drag * dt;
        m_velocity.x = std::min(m_velocity.x, -Config::PARASITE_MAX_SPEED);
    }
}

    }

bool Scientist::IsBlockedAhead(const GameContext& ctx, int dir) const
{
    const float tileSize = static_cast<float>(ctx.levelMap->GetTileSize());
    sf::FloatRect bounds = GetBounds();

    auto isSolidAt = [&](float x, float y)
    {
        int tx = static_cast<int>(std::floor(x / tileSize));
        int ty = static_cast<int>(std::floor(y / tileSize));
        return ctx.levelMap->IsSolid(tx, ty, Config::EntityType::Human);
    };

    float aheadX = dir > 0 ? bounds.left + bounds.width + 4.f
                           : bounds.left - 4.f;
    bool wallAhead   = isSolidAt(aheadX, bounds.top + bounds.height * 0.5f);
    bool groundAhead = isSolidAt(aheadX, bounds.top + bounds.height + 4.f);

    return wallAhead || !groundAhead;
}

void StatePlaying::UpdateHUD()
{
    LivingEntity* avatar = m_level.GetAvatar();
    m_floatingHud.SetPosition(avatar->GetPosition());

    sf::String avatarName =
        avatar->GetProperties().GetSfString("name", ToSfStringFromUtf8("Неизвестный"));
    bool isInElevator = m_level.IsAvatarInElevator();
    Host* host = m_level.GetCurrentHost();

    if (host != nullptr && avatar == host)
    {
        Military* military = dynamic_cast<Military*>(host);

        if (military != nullptr)
            m_floatingHud.UpdateAmmo(military->GetAmmo(), Config::MILITARY_AMMO);
        else
            m_floatingHud.HideAmmo();

        if (!isInElevator)
        {
            sf::String hint = ToSfStringFromUtf8("Носитель: ") + avatarName;

            if (military != nullptr)
            {
                if (military->IsReloading())
                {
                    hint += ToSfStringFromUtf8("\nПерезарядка...");
                }

                hint += ToSfStringFromUtf8("\nЛКМ: Огонь | ПКМ: Высвободиться");
            }
            else
            {
                hint += ToSfStringFromUtf8("\nПКМ: Высвободиться");
            }

            m_screenHud.SetHintText(hint);
        }

        m_floatingHud.Update(host->GetDecayTimer(), host->GetMaxDecayTimer(),
                             sf::Color(220, 200, 0));
    }
    else
    {
        if (!isInElevator)
            m_screenHud.SetHintText(ToSfStringFromUtf8("Форма: ") + avatarName +
                                    ToSfStringFromUtf8("\nЛКМ: Вселиться"));
        m_floatingHud.HideAmmo();
        const Parasite& p = m_level.GetParasite();
        m_floatingHud.Update(p.GetLife(), p.GetMaxLife(), sf::Color(200, 0, 0));
    }

    if (isInElevator && avatar->GetProperties().GetInt("access_level", 0) < 1)
    {
        m_screenHud.SetHintText(
            ToSfStringFromUtf8("ДОСТУП ЗАПРЕЩЁН!\nТребуется тело учёного!"));
    }

    m_screenHud.SetTimerText(m_level.GetStats().timeSpent, m_data->window.getSize().x);
}

void StatePlaying::HandleLevelResult()
{
    if (m_endStatePushed)
        return;

    switch (m_level.GetResult())
    {
    case LevelResult::Failed:
        m_endStatePushed = true;
        m_data->manager.AddState(StateRef(new StateGameOver(m_data)), false);
        break;

    case LevelResult::Completed:
        m_endStatePushed = true;
        m_data->stats = m_level.GetStats();   // фикс: статистика теперь и для Victory
        if (m_data->currentLevelIndex + 1 < m_data->levels.size())
            m_data->manager.AddState(StateRef(new StateLevelCompleted(m_data)), true);
        else
            m_data->manager.AddState(StateRef(new StateVictory(m_data)), false);
        break;

    default:
        break;
    }
}

            this->m_states.pop();
        }
        else
        {
            this->m_states.top()->Pause();
        }
    }
    this->m_states.push(std::move(m_newState));
    this->m_states.top()->Init();
    this->m_isAdding = false;    
}

void StateManager::ProcessStateChanges()
{
    if (this->m_isRemoving && !this->m_states.empty())
    {
        ProcessRemoveState();
    }

    if (this->m_isAdding)
    {
        ProcessAddState();
    }
}

StateRef& StateManager::GetActiveState()
{
    return this->m_states.top();
}

void UIMenu::AddButton(const sf::String& text, const sf::Font& font, float yPos, float windowWidth, std::function<void()> onClick)

    m_buttons.emplace_back(text, font, yPos, windowWidth, onClick);

void UIMenu::MenuUp()
{
    if (m_buttons.empty()) return;
    if (m_selectedOption > 0) m_selectedOption--;
    else m_selectedOption = m_buttons.size() - 1;
}

void UIMenu::MenuDown()
{
    if (m_buttons.empty()) return;
    if (m_selectedOption < m_buttons.size() - 1) m_selectedOption++;
    else m_selectedOption = 0;
}

void UIMenu::HandleInput(const sf::Event& event, SoundManager& sound)
{
    if (GameKeyboard::IsEventUp(event)) {
        MenuUp();
        sound.PlaySound(SOUND_ID_BUTTON);
    }
    else if (GameKeyboard::IsEventDown(event)) {
        MenuDown();
        sound.PlaySound(SOUND_ID_BUTTON);
    } 
    else if (GameKeyboard::IsEventOpen(event)) {
        if (!m_buttons.empty()) {
            sound.PlaySound(SOUND_ID_BUTTON);
            m_buttons[m_selectedOption].OnClick();
        }
    }
}

void UIMenu::Update()
{
    for (size_t i = 0; i < m_buttons.size(); ++i) {
        m_buttons[i].SetSelected(i == m_selectedOption);
    }
}

void UIMenu::Draw(sf::RenderWindow& window)
{
    for (auto& btn : m_buttons) {
        btn.Draw(window);
    }
}

void EntityHUD::UpdateAmmo(int current, int maxAmmo)
{
    if (maxAmmo <= 0)
    {
        HideAmmo();
        return;
    }

    m_showAmmo = true;

    if (static_cast<int>(m_ammoDots.size()) != maxAmmo)
    {
        m_ammoDots.resize(maxAmmo);

        for (auto& dot : m_ammoDots)
        {
            dot.setRadius(Config::AMMO_DOT_RADIUS);
            dot.setOrigin(Config::AMMO_DOT_RADIUS, Config::AMMO_DOT_RADIUS);
            dot.setOutlineThickness(1.f);
            dot.setOutlineColor(sf::Color::Black);
        }

        UpdateAmmoPositions();
    }

    for (int i = 0; i < maxAmmo; ++i)
    {
        m_ammoDots[i].setFillColor(i < current ? Config::AMMO_DOT_FILL_COLOR
                                               : Config::AMMO_DOT_EMPTY_COLOR);
    }
}

void EntityHUD::UpdateAmmoPositions()

    if (m_ammoDots.empty())

        m_ammoDots[i].setPosition(startX + i * Config::AMMO_DOT_SPACING, y);

#include "Core/Config.hpp"
#include <algorithm>
#include <cmath>

namespace
{
    constexpr unsigned TEXTURE_SIZE = 256;
    constexpr float INNER_RADIUS = 0.55f;
    constexpr float OUTER_RADIUS = 1.0f;
}

void Vignette::Init(sf::Vector2u windowSize)
{
    sf::Image image;
    image.create(TEXTURE_SIZE, TEXTURE_SIZE, sf::Color::Transparent);

    const float center = TEXTURE_SIZE / 2.f;

    for (unsigned y = 0; y < TEXTURE_SIZE; ++y)
    {
        for (unsigned x = 0; x < TEXTURE_SIZE; ++x)
        {
            float dx = (x + 0.5f - center) / center;
            float dy = (y + 0.5f - center) / center;
            float distance = std::sqrt(dx * dx + dy * dy);

            float t = std::clamp((distance - INNER_RADIUS) / (OUTER_RADIUS - INNER_RADIUS),
                                 0.f, 1.f);
            t = t * t * (3.f - 2.f * t);

            auto alpha = static_cast<sf::Uint8>(t * 255.f);
            image.setPixel(x, y, sf::Color(255, 255, 255, alpha));
        }
    }

    m_texture.loadFromImage(image);
    m_texture.setSmooth(true);

    m_sprite.setTexture(m_texture, true);
    m_sprite.setPosition(0.f, 0.f);
    m_sprite.setScale(windowSize.x / static_cast<float>(TEXTURE_SIZE),
                      windowSize.y / static_cast<float>(TEXTURE_SIZE));
    m_sprite.setColor(sf::Color(Config::VIGNETTE_COLOR.r,
                                Config::VIGNETTE_COLOR.g,
                                Config::VIGNETTE_COLOR.b,
                                0));
}

void Vignette::Update(float dt, bool active)
{
    if (active)
    {
        m_time += dt * Config::HOST_DANGER_PULSE_SPEED;
    }
    else
    {
        m_time = 0.f;
    }

    float pulse = active ? 0.5f + 0.5f * std::sin(m_time) : 0.f;
    float target = active ? Config::VIGNETTE_BASE_ALPHA +
                            Config::VIGNETTE_PULSE_ALPHA * pulse
                          : 0.f;

    m_alpha += (target - m_alpha) * std::min(1.f, dt * 6.f);

    sf::Color color = Config::VIGNETTE_COLOR;
    color.a = static_cast<sf::Uint8>(std::clamp(m_alpha, 0.f, 255.f));
    m_sprite.setColor(color);
}

void Vignette::Draw(sf::RenderWindow& window)
{
    if (m_alpha > 0.f)
    {
        window.draw(m_sprite);
    }
}
