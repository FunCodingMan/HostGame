void AnimationController::Setup(const sf::Texture& tex, int frameW, int frameH)
{
    m_texture = &tex;
    m_frameW  = frameW;
    m_frameH  = frameH;
    m_nextRow = 0;
}

void AnimationController::AddClip(const std::string& name, int frameCount, float frameTime)
{
    if (m_texture == nullptr || frameCount <= 0) return;

    int rows = (int)(m_texture->getSize().y / m_frameH);
    if (m_nextRow >= rows) return;

    int row = m_nextRow++;
    AddClipRow(name, row, 0, frameCount, frameTime);
}

void AnimationController::AddClip(const std::string& name, int frameCount, float frameTime,
                                  int row, int startCol)
{
    if (m_texture == nullptr || frameCount <= 0 || row < 0 || startCol < 0) return;

    int rows = (int)(m_texture->getSize().y / m_frameH);
    if (row >= rows) return;

    AddClipRow(name, row, startCol, frameCount, frameTime);

    if (row + 1 > m_nextRow) m_nextRow = row + 1;
}

void AnimationController::AddClip(const std::string& name,
                                  const std::vector<sf::IntRect>& frames,
                                  float frameTime, bool loop)
{
    m_clips[name] = Clip{ frames, frameTime, loop };
}

void AnimationController::AddClipRow(const std::string& name, int row, int startCol,
                                     int count, float frameTime)
{
    int cols = (int)(m_texture->getSize().x / m_frameW);
    int realCount = std::min(count, cols - startCol);
    if (realCount <= 0) return;

    std::vector<sf::IntRect> frames;
    frames.reserve(realCount);
    for (int i = 0; i < realCount; ++i)
    {
        frames.emplace_back((startCol + i) * m_frameW, row * m_frameH, m_frameW, m_frameH);
    }

    bool  loop = (realCount > 1);
    float ft   = (realCount > 1) ? frameTime : 1.f;

    m_clips[name] = Clip{ frames, ft, loop };
}

void AnimationController::Play(const std::string& name)
{
    if (m_current == name) return;
    if (!m_clips.count(name)) return;
    m_current = name;
    m_frame   = 0;
    m_elapsed = 0.f;
}

void AnimationController::Update(float dt)
{
    auto it = m_clips.find(m_current);
    if (it == m_clips.end() || it->second.frames.empty()) return;

    const Clip& c = it->second;
    m_elapsed += dt;
    while (m_elapsed >= c.frameTime)
    {
        m_elapsed -= c.frameTime;
        if (++m_frame >= (int)c.frames.size())
            m_frame = c.loop ? 0 : (int)c.frames.size() - 1;
    }
}

void AnimationController::Apply(sf::Sprite& sprite) const
{
    auto it = m_clips.find(m_current);
    if (it == m_clips.end() || it->second.frames.empty()) return;
    sprite.setTextureRect(it->second.frames[m_frame]);
}

Camera::Camera(sf::Vector2f size) : m_lerpSpeed(CAMERA_LERP_FACTOR)
{
    m_view.setSize(size);
    m_view.setCenter(size.x / 2.f, size.y / 2.f);
    m_deadZone = sf::Vector2f(CAMERA_DEAD_ZONE_WIDTH, CAMERA_DEAD_ZONE_HEIGHT);

    m_currentFocus = m_view.getCenter();
    m_currentMouseOffset = sf::Vector2f(0.f, 0.f);
}

void Camera::Update(sf::Vector2f targetPos, sf::Vector2f targetMouseOffset, float dt, float mapWidthPx, float mapHeightPx)
{
    sf::Vector2f desiredFocus = m_currentFocus;
    
    if (targetPos.x > m_currentFocus.x + m_deadZone.x) {
        desiredFocus.x = targetPos.x - m_deadZone.x;
    } else if (targetPos.x < m_currentFocus.x - m_deadZone.x) {
        desiredFocus.x = targetPos.x + m_deadZone.x;
    }

    if (targetPos.y > m_currentFocus.y + m_deadZone.y) {
        desiredFocus.y = targetPos.y - m_deadZone.y;
    } else if (targetPos.y < m_currentFocus.y - m_deadZone.y) {
        desiredFocus.y = targetPos.y + m_deadZone.y;
    }

    m_currentFocus += (desiredFocus - m_currentFocus) * m_lerpSpeed * dt;

    m_currentMouseOffset += (targetMouseOffset - m_currentMouseOffset) * m_lerpSpeed * dt;

    sf::Vector2f finalPos = m_currentFocus + m_currentMouseOffset;
    
    float halfWidth = m_view.getSize().x / 2.f;
    float halfHeight = m_view.getSize().y / 2.f;

    finalPos.x = std::clamp(finalPos.x, halfWidth, mapWidthPx - halfWidth);
    finalPos.y = std::clamp(finalPos.y, halfHeight, mapHeightPx - halfHeight);

    m_view.setCenter(finalPos);
}

const sf::View& Camera::GetView() const
{
    return m_view;
}

void Camera::SetSize(sf::Vector2f newSize)
{
    m_view.setSize(newSize);
}

void Camera::Zoom(float factor)
{
    m_view.zoom(factor);
}

Game::Game() 
{
    m_data->window.create(sf::VideoMode::getDesktopMode(), "The Host", sf::Style::Fullscreen);
    m_data->window.setKeyRepeatEnabled(false);
    m_data->window.setVerticalSyncEnabled(true);
    m_data->headerFont.loadFromFile("../assets/fonts/PressStart2P-Regular.ttf");
    m_data->hudFont.loadFromFile("../assets/fonts/Sligoil-Micro.ttf");
    m_data->manager.AddState(StateRef(new StateMainMenu(m_data)));
}

void Game::Run() 
{
    sf::Clock clock;
    float accumulator = 0.0f;

    while (m_data->window.isOpen()) 
    {
        m_data->manager.ProcessStateChanges();

        ProcessEvents();

        float frameTime = clock.restart().asSeconds();
        if (frameTime > 0.25f) frameTime = 0.25f;
        accumulator += frameTime;

        while (accumulator >= m_dt) 
        {
            Update();
            accumulator -= m_dt;
        }

        Render();
    }
}

void Game::ProcessEvents()
{
    sf::Event event;
    while (m_data->window.pollEvent(event)) 
    {
        if (event.type == sf::Event::Closed) {
            m_data->window.close();
        }
        
        if (!m_data->manager.isEmpty()) {
            m_data->manager.GetActiveState()->HandleInput(event);
        }
    }
}

void Game::Update()
{
    if (!m_data->manager.isEmpty()) {
        m_data->manager.GetActiveState()->Update(m_dt);
    }
}

void Game::Render()
{
    m_data->window.clear(Config::BG_COLOR);
    
    if (!m_data->manager.isEmpty()) {
        m_data->manager.GetActiveState()->Draw();
    }

    m_data->window.display();
}

Config::CollisionType Map::StringToColisionType(const string& s)

    if (s == "Ventilation")

        return Config::CollisionType::Breakable;
    }

void Map::ParseTileConfig(const string& code, const string collisionStr, const int col, const int row)
{
    Config::BlockNames blockName = m_stringToBlock[code];

    Config::CollisionType type = StringToColisionType(collisionStr);

    m_tileDictionary[blockName] = {type, sf::IntRect(col * Config::TEXTURE_TILE_SIZE, 
                                                     row * Config::TEXTURE_TILE_SIZE, 
                                                     Config::TEXTURE_TILE_SIZE, 
                                                     Config::TEXTURE_TILE_SIZE)};
}

bool Map::LoadTileConfig(const string& filename)
{
    ifstream file(filename);
    if(!file.is_open()) return false;

    string line;

    while (getline(file, line))
    {
        if (line.empty() || line[0] == '#')
        {
            continue;
        }
        stringstream ss(line);
        string code, collisionStr;

        int col, row;

        if (ss >> code >> collisionStr >> col >> row)
        {
            if (m_stringToBlock.find(code) == m_stringToBlock.end()) {
                std::cerr << "Внимание: Найден неизвестный код '" << code << "' в " << filename << std::endl;
                continue;
            }
            ParseTileConfig(code, collisionStr, col, row);
        }
    }
    return true;
}

void Map::LoadLeveFromLine(const string& line)
{
    stringstream ss(line);
    string tileCode;
    int curLineWidth = 0;

    while (ss >> tileCode)
    {
        if (m_stringToBlock.find(tileCode) == m_stringToBlock.end())
        {
            std::cerr << "Предупреждение: Неизвестный тайл '" << tileCode << "'" << std::endl;
            m_grid.push_back(Config::BlockNames::Empty);
        }
        else
        {
            m_grid.push_back(m_stringToBlock[tileCode]);
        }
        curLineWidth++;
    }
    if (m_width == 0) 
    {
        m_width = curLineWidth;
    }
    m_height++;
}

bool Map::LoadFromFile(const string& filename)
{
    ifstream file(filename);

    if (!file.is_open())
    {
        std::cerr << "Ошибка: Не удалось загрузить карту " << filename << std::endl;
        return false;
    }

    m_grid.clear();

    string line;
    m_height = 0;
    m_width = 0;

    while (getline(file, line))
    {
        if (line.empty())
        {
            continue;
        }

        LoadLeveFromLine(line);
    }

    BuildVertexArray();

    return true;
}

Config::BlockNames Map::GetTile(int x, int y) const
{
    if (y >= m_height)
    {
        return Config::BlockNames::Empty;
    }
    if (x < 0 || x >= m_width || y < 0)
    {
        return Config::BlockNames::BlackWall;
    }

    return m_grid[y * m_width + x];
}

const Config::TileData* Map::GetTileData(Config::BlockNames blockName) const
{
    auto it = m_tileDictionary.find(blockName);
    if (it != m_tileDictionary.end()) {
        return &it->second;
    }
    return nullptr;
}

bool Map::IsSolid(int x, int y, Config::EntityType entityType) const 
{
    Config::BlockNames block = GetTile(x, y);
    const Config::TileData* data = GetTileData(block);
    
    if (!data) return true; 

    Config::CollisionType type = data->collision;

    if (type == Config::CollisionType::Solid || type == Config::CollisionType::Breakable) return true;
    if (type == Config::CollisionType::Empty) return false;
    
    if (type == Config::CollisionType::Ventilation) 
    {
        return entityType != Config::EntityType::Parasite; 
    }

    return false;
}

void Map::BuildVertexElement(int xTile, int yTile)
{
    Config::BlockNames blockName = GetTile(xTile, yTile);

    const Config::TileData* tileData = GetTileData(blockName);
    if (!tileData)
    {
        return;
    }

    int vertexElIdx = (xTile + yTile * m_width) * 4;

    sf::Vertex* quad = &m_Vertices[vertexElIdx];
    float wSize = Config::WORLD_TILE_SIZE;
    float x = xTile * wSize;
    float y = yTile * wSize;

    quad[0].position = sf::Vector2f(x, y);
    quad[1].position = sf::Vector2f(x + wSize, y);
    quad[2].position = sf::Vector2f(x + wSize, y + wSize);
    quad[3].position = sf::Vector2f(x, y + wSize);

    sf::IntRect texRect = tileData->textureRect;
    float tx = texRect.left;
    float ty = texRect.top;
    float tSize = texRect.width;

    quad[0].texCoords = sf::Vector2f(tx, ty);
    quad[1].texCoords = sf::Vector2f(tx + tSize, ty);
    quad[2].texCoords = sf::Vector2f(tx + tSize, ty + tSize);
    quad[3].texCoords = sf::Vector2f(tx, ty + tSize);
}

void Map::BuildVertexArray()
{
    m_Vertices.resize(m_width * m_height * 4);

    for (int i = 0; i < m_width; ++i)
    {
        for (int j = 0; j < m_height; ++j)
        {
            BuildVertexElement(i, j);
        }
    }
}

bool Map::IsIntersectingWith(const sf::FloatRect& bounds, Config::BlockNames targetBlock) const
{
    int minX = std::floor(bounds.left / Config::WORLD_TILE_SIZE);
    int maxX = std::floor((bounds.left + bounds.width) / Config::WORLD_TILE_SIZE);
    int minY = std::floor(bounds.top / Config::WORLD_TILE_SIZE);
    int maxY = std::floor((bounds.top + bounds.height) / Config::WORLD_TILE_SIZE);

    for (int y = minY; y <= maxY; ++y)
    {
        for (int x = minX; x <= maxX; ++x)
        {
            if (GetTile(x, y) == targetBlock)
            {
                return true;
            }
        }
    }
    return false;
}

bool Map::IsBreakable(int x, int y) const
{
    const Config::TileData* data = GetTileData(GetTile(x, y));
    return data && data->collision == Config::CollisionType::Breakable;
}

void Map::BreakTile(int x, int y)
{
    if (x < 0 || x >= m_width || y < 0 || y >= m_height) return;
    if (!IsBreakable(x, y)) return;

    m_grid[y * m_width + x] = Config::BlockNames::Empty;
    BuildVertexElement(x, y);
}

void Map::Draw(sf::RenderWindow& window) const
{
    window.draw(m_Vertices, &m_tileset);
}

bool Character::CheckMoveKeys(float dt)
{
    bool left  = GameKeyboard::IsLeftKeyPressed();
    bool right = GameKeyboard::IsRightKeyPressed();

    if (left && right)
    {
        return false;
    }

    if (left)
    {
        m_velocity.x -= m_accel * dt;
        return true;
    }

    if (right)
    {
        m_velocity.x += m_accel * dt;
        return true;
    }

    return false;
}

void Character::ResolveFriction(float dt)
{
    float f = m_isOnGround ? m_friction : m_friction * 0.3f;

    if (m_velocity.x > 0.f)
    {
        m_velocity.x -= f * dt;

        if (m_velocity.x < 0.f)
        {
            m_velocity.x = 0.f;
        }
    }
    else
    {
        m_velocity.x += f * dt;

        if (m_velocity.x > 0.f)
        {
            m_velocity.x = 0.f;
        }
    }
}

void Character::CheckMaxSpeed()
{
    if (m_velocity.x > m_maxSpeed)
    {
        m_velocity.x = m_maxSpeed;
    }

    if (m_velocity.x < -m_maxSpeed)
    {
        m_velocity.x = -m_maxSpeed;
    }
}

void Character::UpdateHorizontal(float dt)
{
    bool isMoving = CheckMoveKeys(dt);

    if (!isMoving)
    {
        ResolveFriction(dt);
    }

    CheckMaxSpeed();
}

void Character::UpdateVertical(float dt, bool allowInput)
{
    float curGravity = m_gravity;

    if (allowInput && !GameKeyboard::IsJumpKeyPressed() && m_velocity.y < 0.f)
    {
        curGravity *= 2.f;
    }

    m_velocity.y += curGravity * dt;

    if (m_velocity.y > Config::MAX_FALL_SPEED)
    {
        m_velocity.y = Config::MAX_FALL_SPEED;
    }

    if (allowInput && GameKeyboard::IsJumpKeyPressed() && m_isOnGround)
    {
        m_velocity.y = -m_jumpForce;
        m_isOnGround = false;
    }
}

void Character::UpdateMove(float dt)
{
    UpdateHorizontal(dt);
    UpdateVertical(dt, m_isPlayerControlled);
}


void Character::ResolveCollisionsX(int x, const GameContext& ctx)
{
    int tileSize = ctx.levelMap->GetTileSize();
    sf::FloatRect bounds = m_hitbox.getGlobalBounds();

    if (m_velocity.x > 0.f)
    {
        m_hitbox.setPosition(x * tileSize - (bounds.width / 2.f) - Config::EPSILON,
                             m_hitbox.getPosition().y);
        m_velocity.x = 0.f;
    }
    else if (m_velocity.x < 0.f)
    {
        m_hitbox.setPosition((x + 1) * tileSize + (bounds.width / 2.f) + Config::EPSILON,
                             m_hitbox.getPosition().y);
        m_velocity.x = 0.f;
    }
}

void Character::ResolveCollisionsY(int y, const GameContext& ctx)
{
    int tileSize = ctx.levelMap->GetTileSize();
    sf::FloatRect bounds = m_hitbox.getGlobalBounds();

    if (m_velocity.y > 0.f)
    {
        m_hitbox.setPosition(m_hitbox.getPosition().x,
                             y * tileSize - (bounds.height / 2.f) - Config::EPSILON);
        m_velocity.y = 0.f;
        m_isOnGround = true;
        OnLanding();
    }
    else if (m_velocity.y < 0.f)
    {
        m_hitbox.setPosition(m_hitbox.getPosition().x,
                             (y + 1) * tileSize + (bounds.height / 2.f) + Config::EPSILON);
        m_velocity.y = 0.f;
    }
}

void Character::ResolveCollisions(const GameContext& ctx, Axis axis)
{
    sf::FloatRect bounds = m_hitbox.getGlobalBounds();
    int tileSize = ctx.levelMap->GetTileSize();

    int minX = (int)std::floor(bounds.left / tileSize);
    int maxX = (int)std::floor((bounds.left + bounds.width)  / tileSize);
    int minY = (int)std::floor(bounds.top  / tileSize);
    int maxY = (int)std::floor((bounds.top + bounds.height) / tileSize);

    if (axis == Axis::Vertical)
    {
        m_isOnGround = false;
    }

    for (int y = minY; y <= maxY; ++y)
    {
        for (int x = minX; x <= maxX; ++x)
        {
            if (!ctx.levelMap->IsSolid(x, y, GetEntityType()))
            {
                continue;
            }

            if (TryBreakTile(x, y, ctx))
            {
                continue;
            }

            if (axis == Axis::Horizontal)
            {
                ResolveCollisionsX(x, ctx);
            }
            else
            {
                ResolveCollisionsY(y, ctx);
            }
        }
    }
}


void Character::UpdateSpriteDirection()
{
    if (m_velocity.x > 0.f)
    {
        m_sprite.setScale(m_scale, m_scale);
    }
    else if (m_velocity.x < 0.f)
    {
        m_sprite.setScale(-m_scale, m_scale);
    }
}

void Character::UpdateAnimation(float dt)
{
    const char* name = GetAnimationName();

    if (name == nullptr)
    {
        return;
    }

    m_anim.Play(name);
    m_anim.Update(dt);
    m_anim.Apply(m_sprite);
}

void Character::ApplyMovement(const GameContext& ctx)
{
    float dt = ctx.dt;

    m_hitbox.move(m_velocity.x * dt, 0.f);
    ResolveCollisions(ctx, Axis::Horizontal);

    m_hitbox.move(0.f, m_velocity.y * dt);
    ResolveCollisions(ctx, Axis::Vertical);

    UpdateSpriteDirection();
    UpdateAnimation(dt);

    sf::Vector2f eff = m_spriteOffset;

    if (m_sprite.getScale().x < 0.f)
    {
        eff.x = -eff.x;
    }

    m_sprite.setPosition(m_hitbox.getPosition() + eff);
}

void Parasite::ForceDash(sf::Vector2f targetPos)
{
    sf::Vector2f center = m_hitbox.getPosition();

    float dx = targetPos.x - center.x;
    float dy = targetPos.y - center.y;
    float length = std::sqrt(dx * dx + dy * dy);

    if (length == 0.f)
    {
        return;
    }

    m_velocity.x = (dx / length) * m_dashForce;
    m_velocity.y = (dy / length) * m_dashForce;

    m_dashCooldown = Config::DASH_COOLDOWN_TIME;
    m_dashTimer    = Config::DASH_DURATION;
    m_hasDash      = false;
    m_isOnGround   = false;
}

void Parasite::UpdateDash(float dt, sf::Vector2f mousePos)
{
    if (m_dashCooldown > 0.f)
    {
        m_dashCooldown -= dt;
    }

    if (GameKeyboard::IsDashKeyPressed() && m_dashCooldown <= 0.f && m_hasDash)
    {
        ForceDash(mousePos);
    }
}

void Parasite::UpdateDashTimer(float dt)
{
    if (m_dashTimer > 0.f)
    {
        m_dashTimer -= dt;

        if (m_dashTimer <= 0.f)
        {
            m_velocity.x *= 0.5f;
            m_velocity.y *= 0.5f;
        }
    }
}

void Parasite::HorizontalDash(float dt)
{
    float drag = m_friction * 3.f;

    if (m_velocity.x > 0.f)
    {
        m_velocity.x -= drag * dt;
        m_velocity.x = std::max(m_velocity.x, Config::PARASITE_MAX_SPEED);
    }
    else
    {
        m_velocity.x += drag * dt;
        m_velocity.x = std::min(m_velocity.x, -Config::PARASITE_MAX_SPEED);
    }
}

void Parasite::UpdateHorizontal(float dt)
{
    if (std::abs(m_velocity.x) > m_maxSpeed)
    {
        HorizontalDash(dt);
    }
    else
    {
        Character::UpdateHorizontal(dt);
    }
}

void Parasite::UpdateMove(float dt)

    UpdateDashTimer(dt);

    if (m_dashTimer <= 0.f)
    {
        UpdateHorizontal(dt);
        UpdateVertical(dt, true);
    }
}

void StatePlaying::UpdateCamera(float dt)
{
    sf::Vector2f targetPos = m_playerAvatar->GetPosition();
    sf::Vector2i mousePixelPos = sf::Mouse::getPosition(m_data->window);
    sf::Vector2u windowSize = m_data->window.getSize();

    float normX = (mousePixelPos.x - (windowSize.x / 2.f)) / (windowSize.x / 2.f);
    float normY = (mousePixelPos.y - (windowSize.y / 2.f)) / (windowSize.y / 2.f);

    normX = std::clamp(normX, -1.f, 1.f);
    normY = std::clamp(normY, -1.f, 1.f);

    sf::Vector2f viewSize = m_camera.GetView().getSize();
    float maxOffsetX = (viewSize.x / 2.f) * Config::CAMERA_LOOK_STRENGTH;
    float maxOffsetY = (viewSize.y / 2.f) * Config::CAMERA_LOOK_STRENGTH;

    sf::Vector2f mouseOffset(normX * maxOffsetX, normY * maxOffsetY);

    m_camera.Update(targetPos, mouseOffset, dt, m_map.GetWidthInPixels(),
                    m_map.GetHeightInPixels());
}

void StatePlaying::Update(float dt)
{
    UpdateCamera(dt);

    GameContext ctx;
    ctx.dt = dt;
    ctx.mousePos =
        m_data->window.mapPixelToCoords(sf::Mouse::getPosition(m_data->window), m_camera.GetView());
    ctx.levelMap = &m_map;

    m_parasite.Update(ctx);

    for (auto &enemy : m_hosts)

        enemy->Update(ctx);

    HandleEjection(ctx);

}

void StatePlaying::HandlePossession()
{
    if (m_playerAvatar != &m_parasite || !m_parasite.IsDashing())
        return;

    for (auto &enemy : m_hosts)
    {
        if (!enemy->IsDead() && m_parasite.GetBounds().intersects(enemy->GetBounds()))
        {
            m_parasite.SetPlayerControlled(false);

            enemy->SetPlayerControlled(true);
            enemy->ResetDecayTimer();

            m_playerAvatar = enemy.get();
            m_currentHost = enemy.get();
            m_timeInCurrentHost = 0.f;
            break;
        }
    }
}

void StatePlaying::HandleEjection(const GameContext &ctx)
{
    if (m_currentHost == nullptr || m_playerAvatar != m_currentHost)
        return;

    m_timeInCurrentHost += ctx.dt;

    bool manualEject = sf::Mouse::isButtonPressed(sf::Mouse::Right);
    bool timeOut = (m_currentHost->GetDecayTimer() <= 0.f);

    if (manualEject && m_timeInCurrentHost < Config::COOLDOWN_IN_HOST)
    {
        manualEject = false;
    }

    if (manualEject || timeOut)
    {
        m_currentHost->SetPlayerControlled(false);
        m_currentHost->TakeDamage(9999.f);

        m_parasite.SetPosition(m_currentHost->GetPosition());
        m_parasite.SetPlayerControlled(true);

        if (manualEject)
        {
            m_parasite.ForceDash(ctx.mousePos);
        }

        m_playerAvatar = &m_parasite;
        m_currentHost = nullptr;
    }
}

void StatePlaying::UpdateHUD()
{
    m_floatingHud.SetPosition(m_playerAvatar->GetPosition());

    if (m_currentHost != nullptr && m_playerAvatar == m_currentHost)
    {
        m_floatingHud.Update(m_currentHost->GetDecayTimer(), m_currentHost->GetMaxDecayTimer(),
                             sf::Color(220, 200, 0));
    }
    else
    {
        m_floatingHud.Update(m_parasite.GetLife(), m_parasite.GetMaxLife(), sf::Color(200, 0, 0));
    }
}

void StatePlaying::CheckEndGameConditions()
{

    bool isParasiteDead = (m_parasite.GetLife() <= 0.f);

    bool fellIntoPit = (m_playerAvatar->GetPosition().y > m_map.GetHeightInPixels());

    if (isParasiteDead || fellIntoPit)
    {
        m_data->manager.AddState(StateRef(new StateGameOver(m_data)), false);
        return;
    }

    if (m_currentHost != nullptr)
    {
        if (m_map.IsIntersectingWith(m_playerAvatar->GetBounds(), Config::BlockNames::Elevator))
        {
            m_data->manager.AddState(StateRef(new StateVictory(m_data)), false);
            return;
        }
    }

}

{
    this->m_isAdding = true;
    this->m_isReplacing = isReplacing;

void StateManager::RemoveState()
{
    this->m_isRemoving = true;
}

void StateManager::ProcessRemoveState()
{
    this->m_states.pop();
    if (!this->m_states.empty())
    {
        this->m_states.top()->Resume();
    }
    this->m_isRemoving = false;    
}

void StateManager::ProcessAddState()
{
    if (!this->m_states.empty())
    {
        if (this->m_isReplacing)
        {
            this->m_states.pop();
        }
        else
        {
            this->m_states.top()->Pause();
        }
    }
    this->m_states.push(std::move(m_newState));
    this->m_states.top()->Init();
    this->m_isAdding = false;    
}

void StateManager::ProcessStateChanges()
{
    if (this->m_isRemoving && !this->m_states.empty())
    {
        ProcessRemoveState();
    }

    if (this->m_isAdding)
    {
        ProcessAddState();
    }
}

StateRef& StateManager::GetActiveState()
{
    return this->m_states.top();
}

UIMenu::UIMenu() : m_selectedOption(0) {}

void UIMenu::AddButton(const sf::String& text, const sf::Font& font, float yPos, float windowWidth, std::function<void()> onClick)
{
    m_buttons.emplace_back(text, font, yPos, windowWidth, onClick);
}

void UIMenu::MenuUp()
{
    if (m_buttons.empty()) return;
    if (m_selectedOption > 0) m_selectedOption--;
    else m_selectedOption = m_buttons.size() - 1;
}

void UIMenu::MenuDown()
{
    if (m_buttons.empty()) return;
    if (m_selectedOption < m_buttons.size() - 1) m_selectedOption++;
    else m_selectedOption = 0;
}

void UIMenu::HandleInput(const sf::Event& event)
{
    if (GameKeyboard::IsEventUp(event)) {
        MenuUp();
    }
    else if (GameKeyboard::IsEventDown(event)) {
        MenuDown();
    } 
    else if (GameKeyboard::IsEventOpen(event)) {
        if (!m_buttons.empty()) {
            m_buttons[m_selectedOption].OnClick();
        }
    }
}

void UIMenu::Update()
{
    for (size_t i = 0; i < m_buttons.size(); ++i) {
        m_buttons[i].SetSelected(i == m_selectedOption);
    }
}

void UIMenu::Draw(sf::RenderWindow& window)
{
    for (auto& btn : m_buttons) {
        btn.Draw(window);
    }
}

void EntityHUD::Update(float currentValue, float maxValue, sf::Color barColor)
{
    if (maxValue <= 0) return;

    float percentage = std::clamp(currentValue / maxValue, 0.f, 1.f);

    m_fgBar.setSize(sf::Vector2f(m_maxWidth * percentage, 6.f));
    m_fgBar.setFillColor(barColor);
}

    window.draw(m_bgBar);
